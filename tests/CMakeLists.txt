cmake_minimum_required(VERSION 3.13)
project(nixboy_tests LANGUAGES C CXX)

enable_testing()
add_executable(debug-launch-test debug_launch_test.cc ../src/debug_launch.cc)
target_include_directories(debug-launch-test PRIVATE ../src)
target_compile_features(debug-launch-test PRIVATE cxx_std_17)
add_test(NAME debug-launch COMMAND debug-launch-test)

add_executable(nixboy-training-state-test nixboy_training_state_test.cc ../src/nixboy_training_state.cc)
target_include_directories(nixboy-training-state-test PRIVATE ../src)
target_compile_features(nixboy-training-state-test PRIVATE cxx_std_17)
add_test(NAME nixboy-training-state COMMAND nixboy-training-state-test)

# This focused Linux fixture links the production skill evaluator and stubs its
# world dependencies. Section GC avoids linking unrelated skill-use gameplay.
find_path(NIXBOY_TEST_ZLIB_INCLUDE_DIR zlib.h HINTS "${FETCHCONTENT_SOURCE_DIR_ZLIB}")
find_path(NIXBOY_TEST_ZCONF_INCLUDE_DIR zconf.h HINTS "${NIXBOY_TEST_ZLIB_INCLUDE_DIR}")
if(UNIX AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NIXBOY_TEST_ZLIB_INCLUDE_DIR AND NIXBOY_TEST_ZCONF_INCLUDE_DIR)
    add_executable(nixboy-skill-reward-test nixboy_skill_reward_test.cc ../src/skill.cc ../src/nixboy_training_state.cc)
    target_include_directories(nixboy-skill-reward-test PRIVATE ../src "${NIXBOY_TEST_ZLIB_INCLUDE_DIR}" "${NIXBOY_TEST_ZCONF_INCLUDE_DIR}")
    target_compile_features(nixboy-skill-reward-test PRIVATE cxx_std_17)
    target_compile_options(nixboy-skill-reward-test PRIVATE -ffunction-sections -fdata-sections)
    set_property(TARGET nixboy-skill-reward-test APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections")
    add_test(NAME nixboy-skill-reward COMMAND nixboy-skill-reward-test)
endif()

find_package(Threads REQUIRED)
add_executable(nixboy-client-test nixboy_client_test.cc ../src/nixboy_client.cc)
target_include_directories(nixboy-client-test PRIVATE ../src)
target_compile_features(nixboy-client-test PRIVATE cxx_std_17)
target_link_libraries(nixboy-client-test PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(nixboy-client-test PRIVATE ws2_32)
endif()
add_test(NAME nixboy-client COMMAND nixboy-client-test)

# Opt-in live probe; never run Docker as part of the ordinary unit test suite.
add_executable(nixboy-live-probe EXCLUDE_FROM_ALL nixboy_live_probe.cc ../src/nixboy_client.cc)
target_include_directories(nixboy-live-probe PRIVATE ../src)
target_compile_features(nixboy-live-probe PRIVATE cxx_std_17)
target_link_libraries(nixboy-live-probe PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(nixboy-live-probe PRIVATE ws2_32)
endif()

add_subdirectory(../third_party/libvterm ${CMAKE_CURRENT_BINARY_DIR}/libvterm)
add_executable(nixboy-terminal-test nixboy_terminal_tests.cc)
target_link_libraries(nixboy-terminal-test PRIVATE nixboy-vterm)
target_compile_features(nixboy-terminal-test PRIVATE cxx_std_17)
add_test(NAME nixboy-terminal COMMAND nixboy-terminal-test)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_executable(nixboy-client-bridge-test nixboy_client_bridge_test.cc ../src/nixboy_client.cc)
    target_include_directories(nixboy-client-bridge-test PRIVATE ../src)
    target_compile_features(nixboy-client-bridge-test PRIVATE cxx_std_17)
    target_link_libraries(nixboy-client-bridge-test PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(nixboy-client-bridge-test PRIVATE ws2_32)
    endif()
    add_test(NAME nixboy-client-bridge
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/nixboy_client_bridge_test.py
            $<TARGET_FILE:nixboy-client-bridge-test>)
    set_tests_properties(nixboy-client-bridge PROPERTIES TIMEOUT 20)
endif()
